3.1 Introduction to HTML for React Developers

Simple Explanation

HTML (HyperText Markup Language) is the skeleton of every webpage. In React, you'll write HTML-like code called JSX that gets transformed into real HTML. Understanding HTML thoroughly is essential because React components ultimately render HTML elements in the browser.

Think of HTML as the structure of a house - the walls, doors, windows, and rooms. React helps you build these structures dynamically, but you still need to know what each building block does and how to use it properly.

Technical Explanation

HTML provides the structural foundation for web applications. In React, JSX is a syntax extension that allows you to write HTML-like code within JavaScript. This JSX gets compiled to React.createElement() calls, which produce virtual DOM elements that are eventually rendered as real HTML elements.

Key concepts for React developers: - Semantic HTML: Using appropriate elements for meaning - Accessibility: ARIA attributes and proper semantics - Forms: Controlled vs uncontrolled components - Events: Synthetic event system - Attributes: React-specific vs HTML attributes

Professional Explanation

In enterprise React development, HTML knowledge extends beyond basic markup. You need to understand:

  • Document Structure: DOCTYPE, meta tags, semantic elements
  • Accessibility: WCAG compliance, ARIA roles, keyboard navigation
  • SEO: Meta tags, semantic structure, Open Graph
  • Performance: Resource loading, prefetching, preloading
  • Security: CSP, XSS prevention, input sanitization
  • Internationalization: lang attributes, dir attributes
  • Progressive Enhancement: Graceful degradation

// Controlled Input with validation
function ControlledForm() {
    const [formData, setFormData] = useState({
        email: '',
        password: '',
        age: 18,
        terms: false
    });

    const [errors, setErrors] = useState({});

    const handleChange = (e) => {
        const { name, value, type, checked } = e.target;
        const val = type === 'checkbox' ? checked : value;

        setFormData(prev => ({
            ...prev,
            [name]: val
        }));

        // Validate on change
        validateField(name, val);
    };

    const validateField = (name, value) => {
        let error = '';
        switch (name) {
            case 'email':
                if (!value || !value.includes('@')) {
                    error = 'Valid email is required';
                }
                break;
            case 'password':
                if (value.length < 8) {
                    error = 'Password must be at least 8 characters';
                }
                break;
            // ... more validations
        }

        setErrors(prev => ({
            ...prev,
            [name]: error
        }));
    };

    return (
        <form>
            <TextInput
                label="Email"
                name="email"
                type="email"
                value={formData.email}
                onChange={handleChange}
                error={errors.email}
                placeholder="Enter email"
                required
            />
            <TextInput
                label="Password"
                name="password"
                type="password"
                value={formData.password}
                onChange={handleChange}
                error={errors.password}
                minLength={8}
            />
            <TextInput
                label="Age"
                name="age"
                type="number"
                value={formData.age}
                onChange={handleChange}
                min={0}
                max={150}
            />
            <div>
                <label>
                    <input
                        type="checkbox"
                        name="terms"
                        checked={formData.terms}
                        onChange={handleChange}
                    />
                    Accept Terms
                </label>
            </div>
        </form>
    );
}

function RegistrationForm() {
    const [formData, setFormData] = useState({
        name: '',
        email: '',
        password: ''
    });
    const [errors, setErrors] = useState({});
    const [isSubmitting, setIsSubmitting] = useState(false);

    const validate = (data) => {
        const errors = {};
        if (!data.name) errors.name = 'Name is required';
        if (!data.email) errors.email = 'Email is required';
        if (data.email && !data.email.includes('@')) {
            errors.email = 'Email is invalid';
        }
        if (data.password && data.password.length < 8) {
            errors.password = 'Password must be at least 8 characters';
        }
        return errors;
    };

    const handleChange = (e) => {
        const { name, value } = e.target;
        setFormData(prev => ({ ...prev, [name]: value }));

        // Clear error for field being typed
        if (errors[name]) {
            setErrors(prev => ({ ...prev, [name]: '' }));
        }
    };

    const handleSubmit = async (e) => {
        e.preventDefault();

        const validationErrors = validate(formData);
        if (Object.keys(validationErrors).length > 0) {
            setErrors(validationErrors);
            return;
        }

        setIsSubmitting(true);
        try {
            await submitForm(formData);
            // Handle success
            console.log('Form submitted successfully');
        } catch (error) {
            // Handle error
            setErrors({ submit: error.message });
        } finally {
            setIsSubmitting(false);
        }
    };

    return (
        <form onSubmit={handleSubmit} noValidate>
            <div className="form-group">
                <label htmlFor="name">Name:</label>
                <input
                    type="text"
                    id="name"
                    name="name"
                    value={formData.name}
                    onChange={handleChange}
                    className={errors.name ? 'error' : ''}

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        if (type === 'bar') {
            drawBarChart(ctx, data, canvas);
        } else if (type === 'line') {
            drawLineChart(ctx, data, canvas);
        }
    }, [data, type]);

        const barWidth = 60;
        const gap = 20;
        const startX = 50;
        const maxHeight = 200;

        data.forEach((item, index) => {
            const height = (item.value / Math.max(...data.map(d => d.value))) * maxHeight;
            const x = startX + (barWidth + gap) * index;
            const y = canvas.height - height - 50;

            ctx.fillStyle = item.color || '#3498db';

            ctx.fillText(item.label, x + barWidth / 2, canvas.height - 20);

            // Add value
            ctx.fillStyle = '#555';
            ctx.fillText(item.value, x + barWidth / 2, y - 10);
        });

function ValidatedForm() {

    const [errors, setErrors] = useState({});

    const validators = {
        username: (value) => {
            if (!value) return 'Username is required';
            if (value.length < 3) return 'Username must be at least 3 characters';
            if (!/^[a-zA-Z0-9_]+$/.test(value)) {
                return 'Username can only contain letters, numbers, and underscores';
            }
            return '';
        },
        email: (value) => {
            if (!value) return 'Email is required';
            if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
                return 'Please enter a valid email address';
            }
            return '';
        },
        password: (value) => {
            if (!value) return 'Password is required';
            if (value.length < 8) return 'Password must be at least 8 characters';
            if (!/[A-Z]/.test(value)) {
                return 'Password must contain at least one uppercase letter';
            }
            if (!/[a-z]/.test(value)) {
                return 'Password must contain at least one lowercase letter';
            }
            if (!/[0-9]/.test(value)) {
                return 'Password must contain at least one number';
            }
            if (!/[!@#$%^&*]/.test(value)) {
                return 'Password must contain at least one special character';
            }
            return '';
        },
        confirmPassword: (value) => {
            if (!value) return 'Please confirm your password';
            if (value !== formData.password) return 'Passwords do not match';
            return '';
        },
        age: (value) => {
            const num = Number(value);
            if (!value) return 'Age is required';
            if (isNaN(num)) return 'Please enter a valid number';
            if (num < 13) return 'You must be at least 13 years old';
            if (num > 120) return 'Please enter a valid age';
            return '';
        },
        website: (value) => {
            if (!value) return '';
            try {
                new URL(value);
                return '';
            } catch {
                return 'Please enter a valid URL';
            }
        },
        birthDate: (value) => {
            if (!value) return 'Birth date is required';
            const date = new Date(value);
            const today = new Date();
            let age = today.getFullYear() - date.getFullYear();
            const monthDiff = today.getMonth() - date.getMonth();
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < date.getDate())) {
                age--;
            }
            if (age < 13) return 'You must be at least 13 years old';
            return '';
        }

        });

function getPasswordStrength(password) {
    let score = 0;
    if (password.length >= 8) score++;
    if (/[A-Z]/.test(password)) score++;
    if (/[a-z]/.test(password)) score++;
    if (/[0-9]/.test(password)) score++;
    if (/[!@#$%^&*]/.test(password)) score++;

    switch (score) {
        case 0:
        case 1: return 'weak';
        case 2:
        case 3: return 'medium';
        case 4:
        case 5: return 'strong';
        default: return 'weak';
    }
}

// Comprehensive ARIA implementation
function AccessibleMenu({ items, onSelect }) {
    const [activeIndex, setActiveIndex] = useState(0);
    const menuRef = useRef(null);

    useEffect(() => {
        if (menuRef.current) {
            const firstItem = menuRef.current.querySelector('[role="menuitem"]');
            if (firstItem) {
                firstItem.focus();
            }
        }
    }, []);

    const handleKeyDown = (e) => {
        let newIndex = activeIndex;
        switch (e.key) {
            case 'ArrowDown':
                e.preventDefault();
                newIndex = (activeIndex + 1) % items.length;
                break;
            case 'ArrowUp':
                e.preventDefault();
                newIndex = (activeIndex - 1 + items.length) % items.length;
                break;
            case 'Home':
                e.preventDefault();
                newIndex = 0;
                break;
            case 'End':
                e.preventDefault();
                newIndex = items.length - 1;
                break;
            case 'Enter':
            case ' ':
                e.preventDefault();
                onSelect(items[activeIndex]);
                break;
            default:
                return;
        }

        setActiveIndex(newIndex);
        const menuItems = menuRef.current.querySelectorAll('[role="menuitem"]');
        menuItems[newIndex]?.focus();
    };

    return (
        <div
            ref={menuRef}
            role="menu"
            aria-label="Navigation menu"
            onKeyDown={handleKeyDown}
        >
            {items.map((item, index) => (
                <div
                    key={item.id}
                    role="menuitem"
                    tabIndex={index === activeIndex ? 0 : -1}
                    aria-current={index === activeIndex ? 'true' : 'false'}
                    onClick={() => onSelect(item)}
                    onMouseEnter={() => setActiveIndex(index)}
                    className={`menu-item ${index === activeIndex ? 'active' : ''}`}
                >
                    {item.icon && <span aria-hidden="true">{item.icon}</span>}
                    {item.label}
                    {item.shortcut && (
                        <span className="shortcut">{item.shortcut}</span>
                    )}
                </div>
            ))}
        </div>
    );
}

function AccessibleModal({ isOpen, onClose, title, children }) {
    const modalRef = useRef(null);

    useEffect(() => {
        if (isOpen) {
            // Focus trap
            modalRef.current?.focus();
            // Prevent background scroll
            document.body.style.overflow = 'hidden';
        }

        return () => {
            document.body.style.overflow = 'unset';
        };
    }, [isOpen]);

    if (!isOpen) return null;

    return (
        <div
            ref={modalRef}
            role="dialog"
            aria-modal="true"
            aria-labelledby="modal-title"
            className="modal-overlay"
            onClick={(e) => {
                if (e.target === e.currentTarget) onClose();
            }}
        >

                    <h2 id="modal-title">{title}</h2>
                    <button
                        onClick={onClose}

                        className="close-button"

                    </button>

                    {children}
                </div>
                <footer>
